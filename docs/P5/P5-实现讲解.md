# P5 S-Core 实现讲解

## 阅读说明

本文档按 **P5 四个 Task** 组织，每个 Task 先说明目标，再逐行讲解实现代码。

- **Task 1**：ioremap / iounmap 实现
- **Task 2**：E1000 网卡轮询模式驱动（初始化 + 发送 + 接收）
- **Task 3**：Shell 添加 send/recv 命令
- **Task 4**：PLIC 初始化 + 外部中断处理 + 网络中断响应

## 零、P5 与 P4 的关键差异

P5 在 P4 基础上新增了网络子系统，内核 BSS 从 ~40KB 膨胀到 ~300KB（E1000 描述符数组），触发了多个基层 Bug：

| 问题 | 根因 | 修复 |
|------|------|------|
| QEMU TLB 崩溃 | 内核 BSS 超出 TLB 预热范围（128KB→300KB） | `boot.c` 预热扩大到 1MB |
| 内核栈踩踏用户内存 | `INIT_KERNEL_STACK=0xffffffc052000000` 指向用户物理内存 | 改为 `0xffffffc050500000` |
| 嵌套中断崩溃 | `lock_kernel()/unlock_kernel()` 在 `do_scheduler` 中重启中断 | 改为 no-op |
| P4 argv 踩踏 recv 栈 | `id==3` 分支为 rw 写 hex 字符串，P5 中 id==3 是 recv | 精简为最小 argc=1 |

## 一、Task 1：ioremap / iounmap

### 1.1 目标

将设备 MMIO 物理地址映射到内核 IO 虚拟地址空间（`IO_ADDR_START = 0xffffffe000000000`），使内核能通过虚拟地址访问 E1000/PLIC 寄存器。

### 1.2 实现：`kernel/mm/ioremap.c`

```c
void *ioremap(unsigned long phys_addr, unsigned long size)
{
    uintptr_t pa = ROUNDDOWN(phys_addr, PAGE_SIZE);
    uintptr_t end = ROUND(phys_addr + size, PAGE_SIZE);
    uintptr_t va = io_base;              // 从 IO_ADDR_START 开始分配
    PTE *pgdir = (PTE *)pa2kva(PGDIR_PA);

    for (uintptr_t offset = 0; pa + offset < end; offset += PAGE_SIZE) {
        uintptr_t v = va + offset;       // 当前虚拟页
        uintptr_t p = pa + offset;       // 当前物理页

        // Sv39 三级页表索引
        uint64_t vpn2 = v >> (12 + 9 + 9);
        uint64_t vpn1 = (v >> (12 + 9)) & 0x1FF;
        uint64_t vpn0 = (v >> 12) & 0x1FF;

        // 按需分配中间页目录
        if (!(pgdir[vpn2] & _PAGE_PRESENT)) {
            uintptr_t t = allocPage(1);
            memset((void *)t, 0, PAGE_SIZE);
            set_pfn(&pgdir[vpn2], kva2pa(t) >> 12);
            set_attribute(&pgdir[vpn2], _PAGE_PRESENT);
        }
        PTE *l1 = (PTE *)pa2kva(get_pa(pgdir[vpn2]));
        if (!(l1[vpn1] & _PAGE_PRESENT)) {
            uintptr_t t = allocPage(1);
            memset((void *)t, 0, PAGE_SIZE);
            set_pfn(&l1[vpn1], kva2pa(t) >> 12);
            set_attribute(&l1[vpn1], _PAGE_PRESENT);
        }

        // 末级页表：直接映射 IO 物理地址（不分配新物理页！）
        PTE *l0 = (PTE *)pa2kva(get_pa(l1[vpn1]));
        set_pfn(&l0[vpn0], p >> 12);
        set_attribute(&l0[vpn0],
            _PAGE_PRESENT | _PAGE_READ | _PAGE_WRITE |
            _PAGE_ACCESSED | _PAGE_DIRTY);
        // 注意：没有 _PAGE_USER！IO 地址仅内核可访问
        local_flush_tlb_page(v);
    }

    io_base = va + (end - pa);    // 推进 IO 虚拟地址分配器
    return (void *)(va + (phys_addr & (PAGE_SIZE - 1)));
}
```

**关键设计决策：**
- `alloc_page_helper` 是为**用户页**设计的（会 `allocPage` 分配物理页 + 设置 `_PAGE_USER`）。IO 映射需要的是将**已有物理地址**写入页表，不分配物理页。所以直接操作 `pgdir`/`l1`/`l0` PTE。
- 不设 `_PAGE_USER`：IO 寄存器不能让用户态访问。

### 1.3 当前使用的简化方案

由于 P5 的 `setup_vm()` 已将 E1000（0x60000000）和 PLIC（0xc000000）身份映射，`init/main.c` 当前直接使用物理地址：

```c
e1000 = (volatile uint8_t *)e1000_phys;       // 身份映射
plic_addr = (uint32_t)bios_read_fdt(PLIC_ADDR);  // 同样
```

`ioremap.c` 中的完整实现留作参考，在需要映射非身份映射的 IO 设备时使用。

## 二、Task 2：E1000 网卡驱动

### 2.1 硬件寄存器操作

```c
// drivers/e1000.h
void e1000_write_reg(volatile uint8_t *base, uint32_t reg, uint32_t value) {
    writel(value, (uint8_t *)base + reg);
}
uint32_t e1000_read_reg(volatile uint8_t *base, uint32_t reg) {
    return readl((uint8_t *)base + reg);
}
void e1000_write_reg_array(volatile uint8_t *base, uint32_t reg, int idx, uint32_t val) {
    writel(val, (uint8_t *)base + reg + (idx << 2));
}
```

`writel`/`readl` 展开为 `*(volatile uint32_t *)(addr) = val` / `*(volatile uint32_t *)(addr)`，确保编译器不优化 MMIO 操作。

### 2.2 关键寄存器

```c
#define E1000_STATUS   0x00008   // 设备状态（bit 1 = Link Up）
#define E1000_RCTL     0x00100   // 接收控制
#define E1000_TCTL     0x00400   // 发送控制
#define E1000_RDBAL    0x02800   // 接收描述符基址低 32 位
#define E1000_RDH      0x02810   // 接收描述符 Head
#define E1000_RDT      0x02818   // 接收描述符 Tail
#define E1000_TDBAL    0x03800   // 发送描述符基址
#define E1000_TDT      0x03818   // 发送描述符 Tail
#define E1000_RA       0x05400   // Receive Address 寄存器数组
#define E1000_IMS      0x000D0   // 中断掩码设置
#define E1000_GPRC     0x04074   // Good Packets Received Count
```

### 2.3 静态数据结构

```c
// 发送描述符（16 字节，16 字节对齐）
struct e1000_tx_desc {
    uint64_t addr;       // 包数据物理地址
    uint16_t length;     // 包长度
    uint8_t  cso;        // Checksum Offset
    uint8_t  cmd;        // 命令（EOP, IFCS, RS）
    uint8_t  status;     // 状态（bit 0 = DD, Descriptor Done）
    uint8_t  css;        // Checksum Start
    uint16_t special;
};

// 接收描述符（16 字节）
struct e1000_rx_desc {
    uint64_t addr;       // 缓冲区物理地址
    uint16_t length;     // 收到包的长度
    uint16_t csum;       // 校验和
    uint8_t  status;     // 状态（bit 0 = DD）
    uint8_t  errors;     // 错误标志
    uint16_t special;
};

#define TXDESCS   64
#define RXDESCS   64
#define TX_PKT_SIZE 2048
#define RX_PKT_SIZE 2048

static struct e1000_tx_desc tx_desc_array[64];
static struct e1000_rx_desc rx_desc_array[64];
static char tx_pkt_buffer[64][2048];   // 发送包缓冲区
static char rx_pkt_buffer[64][2048];   // 接收包缓冲区
```

### 2.4 MAC 地址配置

```c
// 与 pktRxTx 对端 MAC 一致
static const uint8_t enetaddr[6] = {0x00, 0x10, 0x53, 0x00, 0x30, 0x83};

// 写入 Receive Address Register 0
ral = enetaddr[0] | (enetaddr[1] << 8) | (enetaddr[2] << 16) | (enetaddr[3] << 24);
e1000_write_reg_array(e1000, E1000_RA, 0, ral);     // RAL[0] = 低 32 位
rah = enetaddr[4] | (enetaddr[5] << 8) | E1000_RAH_AV;  // AV = Address Valid
e1000_write_reg_array(e1000, E1000_RA, 1, rah);     // RAH[0] = 高 16 位 + AV
```

### 2.5 e1000_init() — 初始化流程

```c
void e1000_init(void)
{
    // ① 探测设备是否存在
    uint32_t probe = e1000_read_reg(e1000, E1000_STATUS);
    if (probe == 0xFFFFFFFF || probe == 0)
        return;   // 无设备，安全退出（make run 时走此路径）

    // ② 重置 E1000：关 RX/TX，清零描述符指针，屏蔽所有中断
    e1000_reset();

    // ③ 配置发送单元
    e1000_configure_tx();

    // ④ 配置接收单元（MAC + 描述符 + RCTL）
    e1000_configure_rx();

    // ⑤ 状态确认打印
    printk("> [E1000] STATUS=0x%x RCTL=0x%x RAL0=0x%x\n", ...);

    // ⑥ 启用接收中断（部分 QEMU e1000 模型要求 IMS 才处理收包）
    e1000_write_reg(e1000, E1000_IMS,
        E1000_IMS_RXT0 | E1000_IMS_RXDMT0);
}
```

**探测机制**：`make run`（无 `-device e1000`）时 `STATUS` 返回 `0xFFFFFFFF`，驱动自动跳过初始化，不会崩溃。

### 2.6 e1000_reset() — 重置设备

```c
static void e1000_reset(void) {
    e1000_write_reg(e1000, E1000_RCTL, 0);      // 关接收
    e1000_write_reg(e1000, E1000_TCTL, 0);      // 关发送
    e1000_write_reg(e1000, E1000_TDH, 0);       // 清零 TX Head
    e1000_write_reg(e1000, E1000_TDT, 0);       // 清零 TX Tail
    e1000_write_reg(e1000, E1000_RDH, 0);       // 清零 RX Head
    e1000_write_reg(e1000, E1000_RDT, 0);       // 清零 RX Tail
    latency(1);                                  // 等待 PCI 事务完成
    e1000_write_reg(e1000, E1000_IMC, 0xffffffff);  // 屏蔽所有中断
    while (0 != e1000_read_reg(e1000, E1000_ICR));  // 清除待处理中断
    tx_tail = 0;
    rx_tail = 0;
}
```

### 2.7 e1000_configure_tx() — 配置发送描述符环

```c
static void e1000_configure_tx(void) {
    // 初始化 64 个发送描述符，指向 tx_pkt_buffer
    for (i = 0; i < TXDESCS; i++) {
        tx_desc_array[i].addr = kva2pa((uintptr_t)tx_pkt_buffer[i]);
        tx_desc_array[i].length = 0;
        tx_desc_array[i].cmd    = 0;
        tx_desc_array[i].status = 0;
    }

    // 告诉硬件描述符环的物理地址和长度
    e1000_write_reg(e1000, E1000_TDBAL, (uint32_t)kva2pa((uintptr_t)tx_desc_array));
    e1000_write_reg(e1000, E1000_TDBAH, (uint32_t)(kva2pa((uintptr_t)tx_desc_array) >> 32));
    e1000_write_reg(e1000, E1000_TDLEN, TXDESCS * sizeof(struct e1000_tx_desc));

    e1000_write_reg(e1000, E1000_TDH, 0);
    e1000_write_reg(e1000, E1000_TDT, 0);

    // 启用发送器
    e1000_write_reg(e1000, E1000_TCTL,
        E1000_TCTL_EN | E1000_TCTL_PSP | (0x10 << 4) | (0x40 << 12));

    // 设置帧间隔
    e1000_write_reg(e1000, E1000_TIPG, ...);
}
```

### 2.8 e1000_configure_rx() — 配置接收描述符环

```c
static void e1000_configure_rx(void) {
    // 设置 MAC 地址（RAL[0] + RAH[0]）
    ral = enetaddr[0] | (enetaddr[1] << 8) | ...;
    rah = enetaddr[4] | (enetaddr[5] << 8) | E1000_RAH_AV;

    // 初始化 64 个接收描述符
    for (i = 0; i < RXDESCS; i++) {
        rx_desc_array[i].addr = kva2pa((uintptr_t)rx_pkt_buffer[i]);
        rx_desc_array[i].status = 0;   // 硬件在收包后设置 DD 位
    }

    // 告知硬件描述符环位置
    e1000_write_reg(e1000, E1000_RDBAL, (uint32_t)kva2pa((uintptr_t)rx_desc_array));
    e1000_write_reg(e1000, E1000_RDBAH, (uint32_t)(kva2pa((uintptr_t)rx_desc_array) >> 32));
    e1000_write_reg(e1000, E1000_RDLEN, RXDESCS * sizeof(struct e1000_rx_desc));

    // Head=0，Tail=63：告诉硬件描述符 0~63 都归硬件所有
    e1000_write_reg(e1000, E1000_RDH, 0);
    e1000_write_reg(e1000, E1000_RDT, RXDESCS - 1);
    rx_tail = RXDESCS - 1;

    // 启用接收：EN | BAM(广播) | UPE(单播混杂) | 2048字节缓冲
    e1000_write_reg(e1000, E1000_RCTL,
        E1000_RCTL_EN | E1000_RCTL_BAM | E1000_RCTL_UPE | E1000_RCTL_SZ_2048);
}
```

**描述符环的拥有权模型：**
```
初始状态：RDT=63 → 硬件拥有描述符 [0, 63]
收包后：硬件写描述符 0（设置 DD、写入长度），RDH 自动推进至 1
驱动轮询：检查 tail+1 的描述符 DD 位 → 读数据 → 更新 RDT 归还描述符
```

**UPE (Unicast Promiscuous)**：必须开启。pktRxTx 发送的目标 MAC (`00:10:53:00:30:83`) 与我们 E1000 MAC 相同，但网络环境可能有杂包。UPE 确保所有单播包都被接收。

### 2.9 e1000_transmit() — 发送数据包

```c
int e1000_transmit(void *txpacket, int length)
{
    if (length <= 0 || length > TX_PKT_SIZE) return 0;

    // ① 从硬件读取当前 Tail 指针
    tail = e1000_read_reg(e1000, E1000_TDT);

    // ② 检查该描述符是否已发送完成（DD 位）
    if (!(tx_desc_array[tail].status & E1000_TXD_STAT_DD)) {
        // 等待完成，超时返回 0
        uint64_t timeout = get_ticks() + 100000;
        while (!(tx_desc_array[tail].status & E1000_TXD_STAT_DD))
            if (get_ticks() > timeout) return 0;
    }

    // ③ 拷贝包数据到发送缓冲区
    memcpy(tx_pkt_buffer[tail], txpacket, length);

    // ④ 配置描述符：设置长度、EOP(包结束)、IFCS(自动 CRC)、RS(报告状态)
    tx_desc_array[tail].length = (uint16_t)length;
    tx_desc_array[tail].cmd    = E1000_TXD_CMD_EOP | E1000_TXD_CMD_IFCS
                               | E1000_TXD_CMD_RS;
    tx_desc_array[tail].status = 0;

    // ⑤ 推进硬件 Tail 指针（硬件开始发送）
    tail = (tail + 1) % TXDESCS;
    e1000_write_reg(e1000, E1000_TDT, tail);

    return length;
}
```

**DD 位等待**：E1000 发送完成后会置位 DD。如果描述符还在发送中（DD=0），需要等待。超时保护防止死循环。

### 2.10 e1000_poll() — 轮询接收数据包

```c
int e1000_poll(void *rxbuffer)
{
    tail = rx_tail;
    next = (tail + 1) % RXDESCS;

    // 检查硬件是否填好了下一个描述符（DD 位）
    if (!(rx_desc_array[next].status & E1000_RXD_STAT_DD))
        return 0;   // 没有新包

    // 获取包长度并拷贝数据
    uint16_t length = rx_desc_array[next].length;
    if (length > 0 && length <= RX_PKT_SIZE)
        memcpy(rxbuffer, rx_pkt_buffer[next], length);

    // 归还描述符给硬件
    rx_desc_array[next].status = 0;
    rx_tail = next;
    e1000_write_reg(e1000, E1000_RDT, next);

    return (int)length;
}
```

**轮询逻辑**：每次都检查 tx_tail+1 位置的 DD 位。硬件按顺序填描述符（0→1→2→...→63→0），驱动按顺序消费。

```c
// tiny_libc/syscall.c
int sys_net_send(void *txpacket, int length) {
    return invoke_syscall(SYSCALL_NET_SEND, (long)txpacket, length, IGNORE, IGNORE, IGNORE);
}
int sys_net_recv(void *rxbuffer, int pkt_num, int *pkt_lens) {
    return invoke_syscall(SYSCALL_NET_RECV, (long)rxbuffer, pkt_num, (long)pkt_lens, IGNORE, IGNORE);
}

// kernel/net/net.c — 内核侧
int do_net_send(void *txpacket, int length) {
    return e1000_transmit(txpacket, length);
}
int do_net_recv(void *rxbuffer, int pkt_num, int *pkt_lens) {
    int total = 0;
    // 先清零 length 数组，防止短读时残留垃圾数据
    if (pkt_lens)
        for (int i = 0; i < pkt_num; i++) pkt_lens[i] = 0;
    // 轮询接收
    for (int i = 0; i < pkt_num; i++) {
        int len = e1000_poll(rxbuffer + total);
        if (len <= 0) break;
        if (pkt_lens) pkt_lens[i] = len;
        total += len;
    }
    return total;   // 返回实际收到的字节数
}
```

## 四、Task 4：PLIC 初始化 + 外部中断处理

### 4.1 PLIC 初始化：`drivers/plic.c`

```c
void plic_init(void)
{
    // ① 找到 E1000 对应的 IRQ 号
    int i;
    for (i = 1; i < nr_irqs; i++) {
        if (plic_read(PLIC_PRIORITY(i)))     // 检查哪些中断有默认优先级
            printk("> [PLIC] irq %d prio=%d\n", i, ...);
    }

    // ② 设置 E1000 中断优先级（QEMU virt 平台 E1000 IRQ = 0x2B = 43）
    nr_irqs = bios_read_fdt(NR_IRQS);
    e1000_irq = PLIC_E1000_QEMU_IRQ;   // = 0x2B = 43
    plic_write(PLIC_PRIORITY(e1000_irq), 1);   // 优先级 = 1

    // ③ 为当前 hart 启用 E1000 中断
    uint32_t mask = plic_read(PLIC_ENABLE(hart, 0));
    mask |= (1u << (e1000_irq % 32));
    plic_write(PLIC_ENABLE(hart, 0), mask);

    // ④ 设置中断阈值（0 = 接受所有优先级的中断）
    plic_write(PLIC_THRESHOLD(hart), 0);
}
```

### 4.2 外部中断入口：`kernel/irq/irq.c` — `handle_irq_ext()`

在 `interrupt_helper()` 中，当 `scause == IRQC_S_EXT`（Supervisor External Interrupt）时调用：

```c
void handle_irq_ext(regs_context_t *regs, uint64_t stval, uint64_t scause)
{
    // ① 从 PLIC 认领中断
    uint32_t hwirq = plic_claim();

    // ② 根据 IRQ 号分发
    if (hwirq == PLIC_E1000_QEMU_IRQ) {
        net_handle_irq();           // E1000 网络中断
    } else if (hwirq != 0) {
        printk("> [IRQ] Unknown external interrupt: %d\n", hwirq);
    }

    // ③ 通知 PLIC 中断处理完成
    if (hwirq != 0)
        plic_complete(hwirq);
}
```

**PLIC 三步骤**：
1. `plic_claim()` — 读 claim 寄存器，获取当前最高优先级中断的 IRQ 号
2. 处理中断
3. `plic_complete(hwirq)` — 写 complete 寄存器，通知 PLIC 中断处理完毕

### 4.3 网络中断处理：`kernel/net/net.c` — `net_handle_irq()`

```c
void net_handle_irq(void)
{
    // ① 读 ICR（Interrupt Cause Read）清除中断原因
    uint32_t icr = e1000_read_reg(e1000, E1000_ICR);

    // ② RX 中断：唤醒等待接收的进程
    if (icr & (E1000_ICR_RXT0 | E1000_ICR_RXDMT0)) {
        while (!list_empty(&recv_block_queue)) {
            do_unblock(recv_block_queue.next);
        }
    }

    // ③ TX 中断：唤醒等待发送的进程
    if (icr & E1000_ICR_TXDW) {
        while (!list_empty(&send_block_queue)) {
            do_unblock(send_block_queue.next);
        }
    }

    // ④ 链路状态变化
    if (icr & E1000_ICR_LSC) {
        printk("> [NET] Link status changed.\n");
    }
}
```

**ICR 读即清除**：读 ICR 寄存器会同时清除所有已触发的中断标志位，无需额外的清除操作。

**阻塞队列**：`recv_block_queue` 和 `send_block_queue` 用于阻塞式网络 IO。当前为轮询模式，队列为空，中断处理仅做 ICR 清除。

### 4.4 中断注册（`init/main.c`）

```c
// 系统调用注册
syscall[SYSCALL_NET_SEND] = (long (*)())do_net_send;
syscall[SYSCALL_NET_RECV] = (long (*)())do_net_recv;

// PLIC 初始化（在 enable_interrupt 前）
plic_init();
e1000_init();

// 启用定时器和中断
bios_set_timer(get_ticks() + TIMER_INTERVAL);
enable_interrupt();    // 设置 sstatus.SIE → PLIC 中断可传递到 CPU
enable_preempt();
do_scheduler();
```

## 八、Design Review 问答

### Q1: 说明发送和接收描述符的格式

#### 发送描述符（TX Descriptor）

```c
// 16 字节，必须 16 字节对齐
struct e1000_tx_desc {
    uint64_t addr;       // [0:7]   包数据缓冲区物理地址
    uint16_t length;     // [8:9]   待发送数据长度（0~2048）
    uint8_t  cso;        // [10]    TCP/UDP Checksum Offset（未使用）
    uint8_t  cmd;        // [11]    命令位
                         //   bit 0: EOP  (End Of Packet，最后一个描述符)
                         //   bit 1: IFCS (Insert FCS，硬件自动加 CRC32)
                         //   bit 3: RS   (Report Status，发送完成后写 DD)
    uint8_t  status;     // [12]    状态位
                         //   bit 0: DD   (Descriptor Done，硬件置 1)
    uint8_t  css;        // [13]    Checksum Start（未使用）
    uint16_t special;    // [14:15] 特殊功能（未使用）
} __attribute__((aligned(16)));
```

**各字段作用：**
- `addr`：存放待发送数据包的**物理地址**。数据先被 `memcpy` 到 `tx_pkt_buffer[i]`，然后其物理地址填入此字段
- `length`：实际有效数据长度，不含 CRC（硬件自动追加）
- `cmd.EOP`：标记这是该帧的最后一个描述符（P5 每帧只用一个描述符，所以始终置位）
- `cmd.IFCS`：告诉硬件自动插入 FCS（CRC32）尾部
- `cmd.RS`：告诉硬件发送完成后写回 DD 状态位
- `status.DD`：硬件发送完成后置为 1，驱动通过此位判断发送是否完成

**发送描述符数组：**
```c
#define TXDESCS 64
static struct e1000_tx_desc tx_desc_array[64];  // 64 个描述符形成环形队列
static char tx_pkt_buffer[64][2048];            // 每个描述符对应一个 2KB 缓冲区
```

#### 接收描述符（RX Descriptor）

```c
// 16 字节
struct e1000_rx_desc {
    uint64_t addr;       // [0:7]   接收缓冲区物理地址
    uint16_t length;     // [8:9]   硬件写入：收到包的实际长度
    uint16_t csum;       // [10:11] 硬件写入：校验和结果
    uint8_t  status;     // [12]    状态位
                         //   bit 0: DD   (Descriptor Done，收到包后硬件置 1)
                         //   bit 1: EOP  (End Of Packet)
    uint8_t  errors;     // [13]    错误标志（CRC错/符号错/序列错等）
    uint16_t special;    // [14:15] VLAN 标签等信息
} __attribute__((aligned(16)));
```

**各字段作用：**
- `addr`：存放接收缓冲区的**物理地址**。驱动初始化时填入 `rx_pkt_buffer[i]` 的物理地址
- `length`：**硬件写回**，表示实际收到的数据长度
- `status.DD`：**硬件置 1**，驱动轮询此位判断是否有新包
- `errors`：硬件设置，用于调试丢包原因

**接收描述符数组：**
```c
#define RXDESCS 64
static struct e1000_rx_desc rx_desc_array[64];  // 64 个描述符形成环形队列
static char rx_pkt_buffer[64][2048];            // 每个描述符对应一个 2KB 缓冲区
```

#### 描述符环的拥有权管理

```
环形队列: desc[0] → desc[1] → ... → desc[63] → desc[0] → ...

初始化时:
  e1000_write_reg(e1000, E1000_RDT, 63);   // Tail = 63
  e1000_write_reg(e1000, E1000_RDH, 0);    // Head = 0
  // 描述符 0~63 全部归硬件所有（tail 之后的全部 + 0 追上 head？）
  // 实际: head=0, tail=63 → 硬件拥有 [0, 63]

收到一个包后:
  硬件: 填描述符 0（DMA 写数据 → 设置 DD → 写 length） → RDH 自动推进到 1
  驱动: 读 rx_tail(63)+1 = 64%64 = 0 → 检查 DD=1 → 拷贝数据
        → 置 status=0 → rx_tail=0 → 写 RDT=0 归还描述符 0

最终:
  RDH = 下一个将要被写的描述符索引
  RDT = 最后一个被驱动消费的描述符索引
  硬件拥有: (RDT+1) 到 (RDH-1) wrapping
  驱动拥有: (RDH) 到 (RDT) wrapping
```

### Q2: 说明从网卡初始化到发送使能或接收使能的全过程

#### 整体流程

```
main()
  → plic_init()        // ① PLIC 中断控制器初始化
  → e1000_init()       // ② 网卡初始化

e1000_init():
  ├── 设备探测 (读 STATUS)
  ├── e1000_reset()             // ③ 完全重置
  ├── e1000_configure_tx()      // ④ 配置发送单元 → 发送使能
  ├── e1000_configure_rx()      // ⑤ 配置接收单元 → 接收使能
  └── 写 IMS 使能中断            // ⑥ 中断使能
```

#### ③ e1000_reset()：完全重置

```c
static void e1000_reset(void) {
    e1000_write_reg(e1000, E1000_RCTL, 0);      // ① 禁用接收器
    e1000_write_reg(e1000, E1000_TCTL, 0);      // ② 禁用发送器
    e1000_write_reg(e1000, E1000_TDH, 0);       // ③ 清零 TX Head/Tail
    e1000_write_reg(e1000, E1000_TDT, 0);
    e1000_write_reg(e1000, E1000_RDH, 0);       // ④ 清零 RX Head/Tail
    e1000_write_reg(e1000, E1000_RDT, 0);
    latency(1);                                   // ⑤ 等待 PCI 事务完成
    e1000_write_reg(e1000, E1000_IMC, 0xffffffff); // ⑥ 屏蔽所有中断
    while (e1000_read_reg(e1000, E1000_ICR));     // ⑦ 清除所有待处理中断
    tx_tail = 0; rx_tail = 0;                     // ⑧ 重置软件指针
}
```

这一步将 E1000 恢复到上电默认状态：RCTL/TCTL 清零 → 收发器全部关闭，描述符指针清零，中断全部屏蔽。

#### ④ e1000_configure_tx()：配置发送单元 → 发送使能

```c
static void e1000_configure_tx(void) {
    // 步骤1: 初始化软件描述符
    for (i = 0; i < TXDESCS; i++) {
        tx_desc_array[i].addr   = kva2pa((uintptr_t)tx_pkt_buffer[i]);  // 指向缓冲区
        tx_desc_array[i].length = 0;
        tx_desc_array[i].cmd    = 0;
        tx_desc_array[i].status = 0;   // DD=0，驱动拥有此描述符
    }

    // 步骤2: 告诉硬件描述符环的物理地址
    e1000_write_reg(e1000, E1000_TDBAL, (uint32_t)kva2pa((uintptr_t)tx_desc_array));
    e1000_write_reg(e1000, E1000_TDBAH, (uint32_t)(kva2pa(...) >> 32));
    e1000_write_reg(e1000, E1000_TDLEN, TXDESCS * 16);  // 描述符环总大小

    // 步骤3: 设置 Head/Tail 指针
    e1000_write_reg(e1000, E1000_TDH, 0);
    e1000_write_reg(e1000, E1000_TDT, 0);   // Head=Tail=0，驱动拥有全部描述符

    // 步骤4: 启动物理发送器
    e1000_write_reg(e1000, E1000_TCTL,
        E1000_TCTL_EN |            // 发送使能 ← 关键！
        E1000_TCTL_PSP |           // 短包填充（<64B 自动补 0）
        (0x10 << 4) |              // CT=16（碰撞阈值）
        (0x40 << 12));             // COLD=64（碰撞距离）

    // 步骤5: 设置帧间间隔（IPG）
    e1000_write_reg(e1000, E1000_TIPG,
        DEFAULT_82543_TIPG_IPGT_COPPER |           // IPGT = 10
        (DEFAULT_82543_TIPG_IPGR1 << 10) |          // IPGR1 = 8
        (DEFAULT_82543_TIPG_IPGR2 << 20));          // IPGR2 = 6
}
```

**TCTL.EN = 1 后，发送单元进入 "Idle" 状态，等待 TDT 推进。**

当 `e1000_transmit()` 将数据填入描述符并写 TDT 推进 tail 后，硬件检测到 TDT != TDH → 开始 DMA 读取描述符 → 读取缓冲区数据 → MAC 封帧 → 物理层发送。

#### ⑤ e1000_configure_rx()：配置接收单元 → 接收使能

```c
static void e1000_configure_rx(void) {
    // 步骤1: 设置 MAC 地址（Receive Address Register）
    uint32_t ral = enetaddr[0] | (enetaddr[1] << 8) |
                   (enetaddr[2] << 16) | (enetaddr[3] << 24);
    e1000_write_reg_array(e1000, E1000_RA, 0, ral);         // RAL[0] = 低 32 位

    uint32_t rah = enetaddr[4] | (enetaddr[5] << 8) | E1000_RAH_AV;
    e1000_write_reg_array(e1000, E1000_RA, 1, rah);         // RAH[0] = 高 16 位 + AV

    // AV (Address Valid) = bit 31: 使能此 MAC 地址过滤项

    // 步骤2: 初始化软件描述符
    for (i = 0; i < RXDESCS; i++) {
        rx_desc_array[i].addr   = kva2pa((uintptr_t)rx_pkt_buffer[i]);
        rx_desc_array[i].status = 0;   // DD=0，硬件将在此写入 DD=1
    }

    // 步骤3: 告诉硬件描述符环的物理地址
    e1000_write_reg(e1000, E1000_RDBAL, (uint32_t)kva2pa((uintptr_t)rx_desc_array));
    e1000_write_reg(e1000, E1000_RDBAH, (uint32_t)(kva2pa(...) >> 32));
    e1000_write_reg(e1000, E1000_RDLEN, RXDESCS * 16);

    // 步骤4: 设置 Head/Tail（所有权划分）
    e1000_write_reg(e1000, E1000_RDH, 0);       // Head = 0
    e1000_write_reg(e1000, E1000_RDT, RXDESCS - 1);  // Tail = 63
    rx_tail = RXDESCS - 1;
    // Tail = 63, Head = 0 → 硬件拥有描述符 0~63

    // 步骤5: 启动物理接收器
    e1000_write_reg(e1000, E1000_RCTL,
        E1000_RCTL_EN |             // 接收使能 ← 关键！
        E1000_RCTL_BAM |            // 接受广播帧
        E1000_RCTL_UPE |            // 接受所有单播帧（混杂模式）
        E1000_RCTL_SZ_2048);        // 接收缓冲区 2048 字节
}
```

**RCTL.EN = 1 后，接收单元进入 "Ready" 状态：**
1. 物理层收到以太网帧 → MAC 层校验
2. 检查目标 MAC 是否匹配（RA 寄存器或 BAM/UPE 模式）
3. 匹配 → DMA 写入 `rx_desc[RDH].addr` 指向的缓冲区
4. 写 `length`、设置 `DD` 位 → RDH 自动推进
5. 如果 IMS 中对应的中断使能位被设置 → 触发硬件中断

**接收使能后，即使没有驱动轮询，硬件也会持续接收数据包，直到 64 个描述符填满。** RDH 追上 RDT 时，硬件停止接收（描述符耗尽）。驱动通过 `e1000_poll()` 消费描述符并推进 RDT，归还所有权。

#### ⑥ 中断使能

```c
e1000_write_reg(e1000, E1000_IMS,
    E1000_IMS_RXT0 | E1000_IMS_RXDMT0);
```

- `RXT0`：接收定时器中断（每收到一个包且定时器到期时触发）
- `RXDMT0`：接收描述符最小阈值中断（可用描述符少于阈值时触发）

使能后，P5 的轮询模式仍通过 `e1000_poll()` 检查 DD 位收包，但中断也可触发 `net_handle_irq()` 做 ICR 清除，为未来阻塞式 I/O 做准备。
